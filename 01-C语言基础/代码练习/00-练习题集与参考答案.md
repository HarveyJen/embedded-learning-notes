## 第 01 篇 · C 语言概述与环境搭建（[笔记01](../笔记/01-C语言概述与环境搭建.md)）
- **题 1-1（基础）**：写出 .c 变成可执行文件的四个步骤与各步会报的错，并用 gcc 把 main.c 停在"预处理""编译"两个中间产物。思考提示：预处理只做文本替换不查语法；undefined reference 在哪一步报？
- **题 1-2（嵌入式风格）**：用 printf 打印 `T=25.6C,H=60%`，并把 255 按十进制、十六进制、八进制输出。思考提示：打印 % 本身要连写两个；`%#x` 自动补 0x 前缀。

答案与解析：

```c
/* 题 1-1 四步：-E 只预处理(不查语法)；-S 编译出汇编(语法错误这步报)；-c 汇编出 main.o；链接 undefined reference 报 */
#include <stdio.h>
int main(void)
{
    float temp = 25.6f;                        /* 题 1-2：float 字面量必须加 f */
    printf("T=%.1fC,H=%d%%\n", temp, 60);      /* %% 才能输出 % 本身 */
    printf("dec=%d hex=%#x oct=%#o\n", 255, 255, 255);
    return 0;                                  /* 输出 dec=255 hex=0xff oct=0377 */
}
```
本题考点：题 1-1 考编译四步流水线与报错定位；题 1-2 考 printf 格式符（%.1f、%%、%#x）。
## 第 02 篇 · 数据类型、变量与常量（[笔记02](../笔记/02-数据类型-变量与常量.md)）
- **题 2-1（基础）**：`unsigned char a = 250;` 执行 `a += 10;` 后 a 是多少？`char c = 100; c = c + 100;` 后 c 是多少？写程序验证并解释。思考提示：一个考无符号回绕，一个考整型提升后再截断。
- **题 2-2（嵌入式风格·大小端手算）**：`uint32_t x = 0x12345678;` 存在起始地址 0x20000000，手算小端、大端机上 0x20000000~0x20000003 各字节内容，再写程序判断本机字节序。思考提示：小端 = 低字节放低地址（"个位放门口"）；把地址强转成 `uint8_t *` 逐字节偷看。

答案与解析：

```c
#include <stdio.h>
#include <stdint.h>
int main(void)
{
    unsigned char a = 250;                     /* 题 2-1：250+10=260 超出 0~255 */
    a = (unsigned char)(a + 10);
    printf("a=%u\n", a);                       /* 4：无符号回绕（里程表绕圈） */
    char c = 100;
    c = (char)(c + 100);
    printf("c=%d\n", c);                       /* -56：截断后补码解读 */
    /* 题 2-2 手算：小端(ARM/x86)=78 56 34 12；大端(网络序)=12 34 56 78 */
    uint32_t x = 0x12345678;
    uint8_t *p = (uint8_t *)&x;                /* 强转 1 字节指针逐字节偷看 */
    printf("%02X %02X %02X %02X %s\n", p[0], p[1], p[2], p[3],
           p[0] == 0x78 ? "Little" : "Big");
    return 0;
}
```
本题考点：题 2-1 考无符号回绕与整型提升截断（面试问"unsigned char 存 -1 是多少"答 255）；题 2-2 考大小端定义与指针法判断（联合体法见笔记02 第 8 节）。注：题 2-1 的 c=-56 是 PC gcc 默认 signed char 的结果；ARM GCC 默认 char 为 unsigned，同一程序将打印 200，两者均属正确，以本机编译器实际行为为准。
## 第 03 篇 · 运算符与流程控制（[笔记03](../笔记/03-运算符与流程控制.md)）
- **题 3-1（基础）**：求 1~100 中所有"能被 3 整除但不能被 5 整除"的数之和。思考提示：% 判倍数；continue 跳过本轮。
- **题 3-2（嵌入式风格）**：写"串口菜单"状态机雏形：主菜单收到 '1' 跳 LED 子菜单，LED 态收到 '0' 返回主菜单，收到 'q' 退出，用枚举 + switch 实现。思考提示：状态机 = 枚举状态 + switch 分发，state 是唯一进度条。

答案与解析：

```c
#include <stdio.h>
typedef enum { M_MAIN, M_LED, M_EXIT } State;  /* 枚举当状态用 */
int main(void)
{
    int sum = 0;                               /* 题 3-1 */
    for (int i = 1; i <= 100; i++) {
        if (i % 5 == 0) continue;              /* 被 5 整除：本轮作废 */
        if (i % 3 == 0) sum += i;              /* 被 3 整除：累加 */
    }
    printf("sum=%d\n", sum);                   /* 1368 */
    const char keys[] = { '1', '0', 'q' };     /* 题 3-2：模拟按键，真实来自串口 */
    State s = M_MAIN;
    for (int i = 0; s != M_EXIT; i++) {
        switch (s) {
        case M_MAIN:                           /* 主菜单态：'1' 进子菜单 */
            printf("主菜单 收到'%c'\n", keys[i]);
            s = (keys[i] == '1') ? M_LED : M_EXIT;
            break;
        case M_LED:                            /* LED 态：'0' 返回主菜单 */
            printf("LED菜单 收到'%c'\n", keys[i]);
            s = (keys[i] == '0') ? M_MAIN : M_EXIT;
            break;
        default: s = M_EXIT; break;            /* 兜底，防状态跑飞 */
        }
    }
    return 0;                                  /* 循环条件 state != M_EXIT 负责退出 */
}
```
本题考点：题 3-1 考取余与 continue；题 3-2 考枚举 + switch 状态机骨架（配中断改状态 = 裸机完整框架，见笔记10 第 8 节）。
## 第 04 篇 · 函数（[笔记04](../笔记/04-函数.md)）
- **题 4-1（基础）**：写递归函数求 1+2+...+n，手写展开 my_sum(4)，并回答嵌入式为什么慎用递归。思考提示：递归两要素 = 出口 + 规模递减；MCU 栈只有几 KB。
- **题 4-2（嵌入式风格）**：实现遍历器 `for_each(arr, n, fn)`，传入不同函数指针实现"十进制打印"与"十六进制打印"。思考提示：函数名就是地址；声明格式 `返回类型 (*p)(参数)`。

答案与解析：

```c
#include <stdio.h>
/* 题 4-1：展开 my_sum(4)=4+my_sum(3)=4+(3+(2+my_sum(1)))=4+3+2+1=10。
   慎用递归：每层占一块栈帧，MCU 栈只有几 KB，深了就溢出 */
int my_sum(int n)
{
    if (n == 1) return 1;                      /* 出口：不再往下递 */
    return n + my_sum(n - 1);                  /* 规模递减 */
}
/* 题 4-2：框架固定，行为由函数指针注入（回调=解耦，命令表/中断回调的原型） */
void for_each(const int *arr, int n, void (*fn)(int))
{
    for (int i = 0; i < n; i++) fn(arr[i]);
}
void print_dec(int v) { printf("%d ", v); }
void print_hex(int v) { printf("0x%X ", v); }
int main(void)
{
    printf("sum(4)=%d\n", my_sum(4));          /* 10 */
    int buf[3] = { 10, 20, 30 };
    for_each(buf, 3, print_dec);               /* 10 20 30 */
    for_each(buf, 3, print_hex);               /* 0xA 0x14 0x1E */
    return 0;
}
```
本题考点：题 4-1 考递归两要素与栈溢出风险；题 4-2 考函数指针语法与回调（见笔记10 第 7 节）。
## 第 05 篇 · 数组与字符串（[笔记05](../笔记/05-数组与字符串.md)）
- **题 5-1（基础）**：就地反转字符串 "hello" → "olleh"（不许开新数组）。思考提示：头尾双下标向中间走，边走边交换。
- **题 5-2（嵌入式风格）**：原地删除字符串中所有空格："AT+ CSQ ?" → "AT+CSQ?"（AT 指令清理常用）。思考提示：快慢双下标——快读慢收，收完手动补 '\0'。

答案与解析：

```c
#include <stdio.h>
#include <string.h>
/* 题 5-1：双指针原地逆置 */
void str_reverse(char *s)
{
    int i = 0, j = (int)strlen(s) - 1;         /* j 指向末字符，不含 '\0' */
    while (i < j) {
        char t = s[i]; s[i] = s[j]; s[j] = t;  /* 头尾交换 */
        i++; j--;
    }
}
/* 题 5-2：快慢下标原地去空格 */
void del_space(char *s)
{
    int slow = 0;
    for (int fast = 0; s[fast] != '\0'; fast++)
        if (s[fast] != ' ') s[slow++] = s[fast];   /* 非空格才收 */
    s[slow] = '\0';                            /* 踩坑点：必须补结束符 */
}
int main(void)
{
    char s1[] = "hello";                       /* 必须是数组，字符串常量不可改 */
    str_reverse(s1);
    printf("%s\n", s1);                        /* olleh */
    char s2[] = "AT+ CSQ ?";
    del_space(s2);
    printf("[%s]\n", s2);                      /* [AT+CSQ?] */
    return 0;
}
```
本题考点：题 5-1 考双指针交换模板；题 5-2 考"读写指针"原地修改套路与 '\0' 收尾意识。
## 第 06 篇 · 指针（[笔记06](../笔记/06-指针.md)）
- **题 6-1（基础）**：写出"错误版 swap"与"指针版 swap"及运行结果，并画出 x=1、y=2 调用指针版时的内存示意图。思考提示：想改变谁，就传谁的地址。
- **题 6-2（嵌入式风格·野指针找错）**：指出 `int *make(void) { int a = 10; return &a; }` 的致命问题，给出至少两种修改方案。思考提示：野指针三来源——未初始化、指向已销毁的栈变量、free 后继续用。

答案与解析：

```c
#include <stdio.h>
void swap_wrong(int a, int b)                  /* 题 6-1 错误版：交换的是形参（复印件） */
{ int t = a; a = b; b = t; }
void swap(int *pa, int *pb)                    /* 正确版：传地址改原件 */
{
    int t = *pa; *pa = *pb; *pb = t;
}
int main(void)
{
    int x = 1, y = 2;
    swap_wrong(x, y);
    printf("%d %d\n", x, y);                   /* 1 2 没换成功 */
    swap(&x, &y);
    printf("%d %d\n", x, y);                   /* 2 1 换成功 */
    return 0;
}
/* 内存图（swap 执行中，设 x@0x1000、y@0x1004）：0x1000:x=2  0x1004:y=1
   栈帧:pa=0x1000 pb=0x1004 t=1；*pa=顺门牌号找上门动屋里东西，改的是 x 本尊。
   题 6-2：make 返回栈变量地址，函数返回栈帧销毁 → 野指针。改法：
   ① 变量加 static（挪到数据段）② malloc 到堆（用完 free）③ 调用者传地址进来填 */
```
本题考点：题 6-1 考传值 vs 传址与画内存图习惯；题 6-2 考野指针来源与修复（static / 堆 / 出参）。
## 第 07 篇 · 结构体、联合体、枚举（[笔记07](../笔记/07-结构体-联合体-枚举.md)）
- **题 7-1（基础）**：手算 `struct A { char c; int i; char c2; };` 与 `struct B { char c; char c2; int i; };` 的 sizeof（gcc 规则），写程序验证。思考提示：成员起点对齐自身大小、总大小补到最大对齐数整数倍。
- **题 7-2（嵌入式风格）**：定义 pack(1) 通信帧 {帧头, 长度, 32位数据}，验证 sizeof；再用联合体判断本机大小端。思考提示：pack(1) 让结构与协议帧逐字节一致；联合体成员共享同一块内存。

答案与解析：

```c
#include <stdio.h>
#include <stdint.h>
/* 题 7-1 手算：A=c[0]+填[1..3]+i[4..7]+c2[8]+尾补=12；B=c[0]c2[1]+填[2,3]+i[4..7]=8 */
struct A { char c; int i; char c2; };
struct B { char c; char c2; int i; };
#pragma pack(1)     /* 题 7-2：pack(1) 让结构与协议帧逐字节一致（通信帧标配） */
typedef struct {
    uint8_t  head;                             /* 帧头 0x55 */
    uint8_t  len;                              /* 数据长度 */
    uint32_t value;                            /* 32 位数据 */
} Frame_t;          /* pack(1)=6；不 pack=8（len 后垫 2 字节再放 value） */
#pragma pack()      /* 恢复默认对齐 */
union CheckEndian { /* 联合体成员共享同一块内存 */
    uint16_t u16;
    uint8_t  u8[2];
};
int main(void)
{
    printf("A=%zu B=%zu frame=%zu\n",
           sizeof(struct A), sizeof(struct B), sizeof(Frame_t));    /* 12 8 6 */
    union CheckEndian t;
    t.u16 = 0x1234;
    printf("%s\n", t.u8[0] == 0x34 ? "Little-Endian" : "Big-Endian");
    return 0;
}
```
本题考点：题 7-1 考对齐手算（笔试超高频）与"小成员排前面省空间"；题 7-2 考 pack(1) 用途与联合体判字节序。
## 第 08 篇 · 内存布局与动态内存（[笔记08](../笔记/08-内存布局与动态内存.md)）
- **题 8-1（基础）**：把栈变量、堆内存、未初始化全局变量、static 局部变量的地址打印出来，指出各在哪个分区。思考提示：栈向下增长、堆向上增长；static 局部变量不在栈上。
- **题 8-2（嵌入式风格·找错）**：找出 `int *p = malloc(100); int *q = p; free(p); free(q); *p = 1;` 的 3 个错误并改正。思考提示：q 和 p 是同一块内存的两个名字；口诀"谁租的谁退、退一次就好、退了别再住"。

答案与解析：

```c
#include <stdio.h>
#include <stdlib.h>
int g_zero;                        /* BSS 段：未初始化全局，启动自动清 0 */
int main(void)
{
    int local = 0;                 /* 栈：函数返回即销毁 */
    static int s_local = 1;        /* 数据段：static 局部变量不在栈上！ */
    int *heap = (int *)malloc(4);  /* 堆 */
    /* 题 8-1：栈地址与堆地址相差悬殊，data/bss 在低地址成对出现 */
    printf("stack=%p heap=%p bss=%p data=%p\n",
           (void *)&local, (void *)heap, (void *)&g_zero, (void *)&s_local);
    free(heap);
    heap = NULL;                   /* 防悬空指针 */
    /* 题 8-2 找错：① 没判 NULL；② p、q 同一块内存 free 两次=重复释放；③ free 后再写=悬空。
       正确姿势：判空 → 使用 → free → 置 NULL */
    return 0;
}
```
本题考点：题 8-1 考内存五区对号入座（RAM = data + bss + 堆 + 栈）；题 8-2 考动态内存四步曲与三大经典错误。
## 第 09 篇 · 预处理器与关键字专题（[笔记09](../笔记/09-预处理器与关键字专题.md)）
- **题 9-1（基础·场景判断）**：a) 中断置 1、主循环等待的标志 rx_done；b) 模块内部工具函数 crc16；c) 跨调用累加的计数 cnt；d) 只读但会被硬件改变的状态寄存器 STATUS_REG——各该加什么关键字？思考提示：volatile 管"被外部力量改变"；static 管"生命周期/可见范围"。
- **题 9-2（嵌入式风格）**：`#define SQUARE(x) x * x`，求 SQUARE(1+2) 并解释，写出正确版本，再默写"对变量 v 第 n 位置 1"的宏。思考提示：预处理只做查找替换；宏的参数和整体都要加括号。

答案与解析：

```c
#include <stdio.h>
#include <stdint.h>
static volatile uint8_t rx_done = 0;   /* a: 中断+主循环共享 → volatile（防 -O2 优化成死循环） */
static uint32_t crc16(const uint8_t *p, int len)   /* b: 模块私有函数 → static */
{ (void)p; (void)len; return 0; }
const volatile uint32_t STATUS_REG = 1u;   /* d: 只读但会变 → const+volatile 连用 */
void counter(void)                          /* c: 跨调用累加 → static 局部变量 */
{
    static int cnt = 0;                     /* 只初始化一次，存数据段 */
    printf("cnt=%d\n", ++cnt);
}
#define SQUARE_BAD(x)  x * x                /* 题 9-2 反面教材：参数没括号 */
#define SQUARE(x)      ((x) * (x))          /* 正确版：参数和整体都包住 */
#define SET_BIT(v, n)  ((v) |= (1u << (n))) /* 第 n 位置 1（行业标配写法） */
int main(void)
{
    counter(); counter();                   /* 输出 cnt=1、cnt=2 */
    printf("%d %d %u\n", SQUARE_BAD(1 + 2), SQUARE(1 + 2), STATUS_REG);
    /* BAD 展开成 1+2*1+2=5（文本替换，乘法先算）；SQUARE=(1+2)*(1+2)=9 */
    int v = 0;
    SET_BIT(v, 3);                          /* v = 0000 1000 */
    printf("v=%d\n", v);                    /* 8 */
    (void)crc16; (void)rx_done;             /* 消除"未使用"警告 */
    return 0;
}
```
本题考点：题 9-1 考 volatile（只保证可见性不保证原子性）、static 三用法、const+volatile 连用；题 9-2 考宏替换括号坑与位操作宏（面试必背）。
## 第 10 篇 · 位运算与嵌入式C编程（[笔记10](../笔记/10-位运算与嵌入式C编程.md)）
- **题 10-1（基础·必须闭卷）**：默写对变量 reg 第 n 位"置 1 / 清 0 / 翻转 / 读取"四条语句，并完成"点亮第 n 位"：reg 当 8 位 GPIO 输出寄存器，置位第 2 位、翻转、读取打印，再跑 bit0~bit3 流水灯。思考提示：先用 `1u << n` 造掩码（只有第 n 位是 1 的"挖洞卡片"）。
- **题 10-2（嵌入式风格）**：实现 popcount（统计二进制中 1 的个数）和判断 2 的幂，用 12 和 8 验证。思考提示：`x &= (x-1)` 每次消掉最低位的 1；2 的幂形如 1000...0。

答案与解析：

```c
#include <stdio.h>
#include <stdint.h>
int popcount(uint32_t x)                   /* 题 10-2：统计 1 的个数 */
{
    int n = 0;
    while (x) { x &= (x - 1); n++; }       /* 每循环一次消掉最低位的 1 */
    return n;
}
int is_pow2(uint32_t x)                    /* 2^n: 1000...0 & 0111...1 = 0 */
{
    return x && !(x & (x - 1));
}
int main(void)
{
    uint8_t reg = 0x00;                    /* 题 10-1：当作 8 位 GPIO 输出寄存器 */
    reg |=  (uint8_t)(1u << 2);            /* 置 1：0000 0100 */
    reg &= (uint8_t)~(1u << 2);            /* 清 0 */
    reg ^=  (uint8_t)(1u << 2);            /* 翻转：又亮了 */
    printf("bit2=%u reg=0x%02X\n", (reg >> 2) & 1u, reg);   /* 读取某位 */
    for (int i = 0; i < 4; i++) {          /* 流水灯：每次只亮一个 */
        reg = (uint8_t)(1u << i);
        printf("ODR=0x%02X\n", reg);       /* 01 02 04 08 */
    }
    printf("popcount(12)=%d pow2:8=%d 12=%d\n",
           popcount(12), is_pow2(8), is_pow2(12));          /* 2、1、0 */
    return 0;
}
```

## 第 11 篇 · 链表等基础数据结构（[笔记11](../笔记/11-链表等基础数据结构.md)）
- **题 11-1（基础·必须闭卷）**：手写带头节点单链表的迭代反转（三指针），把 10→20→30 反转成 30→20→10 并打印。思考提示：口诀"先存 next、反向指、头节点接新表头"。
- **题 11-2（嵌入式风格）**：实现环形缓冲 {buf, head, tail} 的 push/pop（中断写 + 主循环读），并解释判满为什么是 `(head+1)%SIZE == tail`。思考提示：牺牲 1 格区分"空"和"满"；各自只写自己的指针。

答案与解析：

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
typedef struct Node { int data; struct Node *next; } Node;
/* 题 11-1：反转三步——先存 next、反向指、头节点接新表头 */
void list_reverse(Node *head)
{
    Node *prev = NULL, *cur = head->next;
    while (cur != NULL) {
        Node *next = cur->next;            /* ① 先存后继，防丢链 */
        cur->next = prev;                  /* ② 反向指 */
        prev = cur; cur = next;            /* ③ 整体后移 */
    }
    head->next = prev;                     /* ④ 哨兵接上新表头 */
}
#define RB_SIZE 8
typedef struct { uint8_t buf[RB_SIZE]; volatile uint8_t head, tail; } RingBuf;
int rb_push(RingBuf *q, uint8_t b)         /* 题 11-2：中断里调用，只写 head */
{
    uint8_t next = (uint8_t)((q->head + 1) % RB_SIZE);
    if (next == q->tail) return -1;        /* 满：留 1 格不用，否则空/满无法区分 */
    q->buf[q->head] = b;                   /* 先放数据，后推进指针（顺序不能反） */
    q->head = next;
    return 0;
}
int rb_pop(RingBuf *q, uint8_t *out)       /* 主循环调用，只写 tail */
{
    if (q->tail == q->head) return -1;     /* 空：两指针重合 */
    *out = q->buf[q->tail];
    q->tail = (uint8_t)((q->tail + 1) % RB_SIZE);
    return 0;
}
int main(void)
{
    Node *head = (Node *)malloc(sizeof(Node)), *tail = head;   /* 哨兵+尾指针 */
    head->next = NULL;                     /* 题解省略判空，工程必须查（笔记08） */
    for (int i = 1; i <= 3; i++) {         /* 尾插建表 10 20 30 */
        Node *n = (Node *)malloc(sizeof(Node));
        n->data = i * 10; n->next = NULL;
        tail->next = n; tail = n;
    }
    list_reverse(head);
    for (Node *c = head->next; c != NULL; c = c->next) printf("%d -> ", c->data);
    printf("NULL\n");                      /* 30 -> 20 -> 10 -> NULL */
    RingBuf rb = { {0}, 0, 0 };
    for (uint8_t i = 1; i <= 10; i++) rb_push(&rb, i);   /* 只装得下 7 个 */
    uint8_t v;
    while (rb_pop(&rb, &v) == 0) printf("%u ", v);       /* 1 2 3 4 5 6 7 */
    return 0;
}
```
