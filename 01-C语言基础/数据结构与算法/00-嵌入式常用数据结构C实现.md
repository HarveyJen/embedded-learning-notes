# 嵌入式常用数据结构 · C 完整实现

> 配套 [笔记11 链表等基础数据结构](../笔记/11-链表等基础数据结构.md) 与 [《C 语言练习题集与参考答案》](../代码练习/00-练习题集与参考答案.md) 使用。五个程序均为完整可编译的独立 .c 文件（类型定义 + 函数实现 + main 测试），复制保存即可运行。

**先说人话**：这五个结构不是应付考试的知识点，是嵌入式代码里天天在岗的"现役零件"，个个有出处——**单链表** = RTOS 就绪任务队列的原型（FreeRTOS list.c，Linux 内核 list_head 同思想）；**环形缓冲区** = 串口收发行业标配（中断收字节塞入，主循环取出解析）；**顺序栈** = 函数调用底层机制 + 协议括号/转义序列配对检查；**循环队列** = 环形缓冲的教科书表兄（RTOS 消息队列雏形）；**哈希命令表** = CLI / AT 指令解析的进阶查表方案。每节按"场景一句话 → 完整代码 → 复杂度与取舍 → 预期输出"展开。

## 一、单链表：增删改查 + 头插法建表 + 反转 + 快慢指针找中点

应用场景一句话：任务挂队、帧排队——数量不定、频繁插删的地方（对应笔记11 第 2~4 节）。

```c
#include <stdio.h>
#include <stdlib.h>
typedef int ElemType;
typedef struct Node {
    ElemType     data;              /* 数据域 */
    struct Node *next;              /* 指针域：自引用必须是指针，不能是本结构体变量 */
} Node;

Node *list_create(void)             /* 带头节点（哨兵）空表：首插/首删不用特判 */
{
    Node *head = (Node *)malloc(sizeof(Node));
    if (head == NULL) return NULL;
    head->data = 0; head->next = NULL;
    return head;
}
int list_push_front(Node *head, ElemType x)   /* 头插建表：O(1)，但顺序是倒的 */
{
    Node *node = (Node *)malloc(sizeof(Node));
    if (node == NULL) return -1;
    node->data = x;
    node->next = head->next;        /* 铁律"先接后断"：顺序反了 → 后半条链丢失 */
    head->next = node;
    return 0;
}
int list_push_back(Node *head, ElemType x)    /* 尾插：保持顺序，要走到底 O(n) */
{
    Node *node = (Node *)malloc(sizeof(Node));
    if (node == NULL) return -1;
    node->data = x; node->next = NULL;
    Node *cur = head;
    while (cur->next != NULL) cur = cur->next;
    cur->next = node;
    return 0;
}
Node *list_find(Node *head, ElemType x)       /* 查：按值找节点，O(n) */
{
    for (Node *cur = head->next; cur != NULL; cur = cur->next)
        if (cur->data == x) return cur;
    return NULL;
}
int list_set(Node *head, ElemType old_v, ElemType new_v)   /* 改 */
{
    Node *pos = list_find(head, old_v);
    if (pos == NULL) return -1;
    pos->data = new_v;
    return 0;
}
int list_remove(Node *head, ElemType x)       /* 删：铁律"先跨过，再 free" */
{
    for (Node *prev = head; prev->next != NULL; prev = prev->next)
        if (prev->next->data == x) {
            Node *victim = prev->next;
            prev->next = victim->next;  /* 前驱跨过它，顺序反了 → 悬空指针 */
            free(victim);
            return 0;
        }
    return -1;
}
void list_reverse(Node *head)       /* 反转（笔试高频）：先存 next、反向指、整体后移 */
{
    Node *prev = NULL, *cur = head->next;
    while (cur != NULL) {
        Node *next = cur->next;
        cur->next = prev;
        prev = cur; cur = next;
    }
    head->next = prev;              /* 哨兵接上新表头 */
}
Node *list_mid(Node *head)          /* 快慢指针找中点：fast 2 步、slow 1 步 */
{
    Node *slow = head->next, *fast = head->next;
    while (fast != NULL && fast->next != NULL) {
        slow = slow->next;
        fast = fast->next->next;
    }
    return slow;                    /* 偶数个节点时取"后半段第一个" */
}
void list_print(const Node *head)
{
    for (const Node *cur = head->next; cur != NULL; cur = cur->next)
        printf("%d -> ", cur->data);
    printf("NULL\n");
}
void list_destroy(Node *head)       /* 销毁：铁律"先存 next，再 free"防泄漏 */
{
    Node *cur = head->next;
    while (cur != NULL) { Node *next = cur->next; free(cur); cur = next; }
    free(head);
}
int main(void)
{
    Node *list = list_create();
    list_push_front(list, 20);      /* 头插建表：顺序会倒 */
    list_push_front(list, 10);      /* 当前：10 20 */
    list_push_back(list, 30); list_push_back(list, 40); list_push_back(list, 50);
    list_print(list);               /* 10 20 30 40 50 */
    list_set(list, 30, 99);         /* 改：30 → 99 */
    list_remove(list, 10);          /* 删头 */
    list_print(list);               /* 20 99 40 50 */
    list_reverse(list);             /* 反转 */
    list_print(list);               /* 50 40 99 20 */
    printf("mid=%d\n", list_mid(list)->data);
    list_destroy(list);
    return 0;
}
```

复杂度与取舍：查找/删改 O(n)（定位后改指针 O(1)），头插 O(1)；每节点多付一个指针的存储。嵌入式取舍：RTOS 用双向循环链表（删除不必找前驱）；裸机里节点常改用静态数组池代替 malloc——少用堆是铁律（笔记08 第 4 节）。

测试预期输出：

```
10 -> 20 -> 30 -> 40 -> 50 -> NULL
20 -> 99 -> 40 -> 50 -> NULL
50 -> 40 -> 99 -> 20 -> NULL
mid=99
```

## 二、环形缓冲区：串口接收场景（重头戏）

应用场景一句话：串口每收 1 字节就在中断里 push，主循环 pop 出来整段解析——中断与主循环之间的"传送带"。

```c
#include <stdio.h>
#include <stdint.h>
#define RB_SIZE 16                   /* 容量取 2 的幂：% 可换成 & (RB_SIZE-1) 更快 */
typedef struct {
    uint8_t  buf[RB_SIZE];
    volatile uint8_t head;           /* 写指针：只有"生产者"（中断）修改 */
    volatile uint8_t tail;           /* 读指针：只有"消费者"（主循环）修改 */
} RingBuf;

int rb_push(RingBuf *q, uint8_t b)   /* 模拟串口接收中断调用——只改 head */
{
    uint8_t next = (uint8_t)((q->head + 1) % RB_SIZE);
    if (next == q->tail) return -1;  /* 判满：留 1 格不用，否则空/满无法区分 */
    q->buf[q->head] = b;             /* 先放数据 */
    q->head = next;                  /* 后推进指针（顺序不能反） */
    return 0;
}
int rb_pop(RingBuf *q, uint8_t *out) /* 主循环调用——只改 tail */
{
    if (q->tail == q->head) return -1;   /* 判空：两指针重合即空 */
    *out = q->buf[q->tail];
    q->tail = (uint8_t)((q->tail + 1) % RB_SIZE);
    return 0;
}
int main(void)
{
    static RingBuf rb;               /* 静态分配：嵌入式尽量不用 malloc */
    int lost = 0, cnt = 0, sum = 0;
    uint8_t v;
    for (int i = 1; i <= 20; i++)    /* 模拟中断连收 20 字节（只能存 15 个） */
        if (rb_push(&rb, (uint8_t)i) != 0) lost++;
    printf("lost=%d\n", lost);       /* 5：满了就丢，绝不覆盖未读数据 */
    while (rb_pop(&rb, &v) == 0) { sum += v; cnt++; }   /* 主循环批量取走 */
    printf("got=%d sum=%d\n", cnt, sum);    /* 15 个，1+2+...+15=120 */
    return 0;
}
```

中断写 + 主循环读为什么不用加锁（单写单读无锁原理）：head 只有中断改、tail 只有主循环改，双方各写各的指针、只读对方的单字节快照；单字节读写在 MCU 上天然原子，配合 volatile 防止编译器把循环里的指针读取优化掉（笔记09 第 7 节），因此无需关中断。边界：若指针/下标是多字节类型跑在 8 位机上，一次读写拆成多条指令，就必须关中断保护。

复杂度与取舍：push/pop 均 O(1)、零动态分配；代价是容量固定、满了只能丢（本例用返回值报告）。容量按"最坏突发数据量 × 主循环处理周期"估算；要求不丢数据就升级成 RTOS 队列（满则阻塞/超时）。

测试预期输出：

```
lost=5
got=15 sum=120
```

## 三、顺序栈：括号匹配（协议转义序列配对）

应用场景一句话：函数调用、表达式求值、"括号/引号/转义序列必须成对"的协议合法性检查——后进先出（像食堂摞盘子）。

```c
#include <stdio.h>
#define STACK_MAX 32
typedef struct {
    char buf[STACK_MAX];
    int  top;                        /* 栈顶下标：-1 表示空 */
} Stack;

void stack_init(Stack *s)  { s->top = -1; }
int  stack_empty(const Stack *s) { return s->top == -1; }

int stack_push(Stack *s, char c)     /* 压栈：先挪 top 再放数 */
{
    if (s->top >= STACK_MAX - 1) return -1;   /* 满 */
    s->buf[++s->top] = c;
    return 0;
}
int stack_pop(Stack *s, char *out)   /* 弹栈：先取数再挪 top */
{
    if (s->top < 0) return -1;                /* 空 */
    *out = s->buf[s->top--];
    return 0;
}
/* 括号匹配：左括号进栈，右括号找栈顶配对（协议查转义序列配对同思路） */
int bracket_match(const char *s)     /* 返回 1 匹配，0 不匹配 */
{
    Stack st; char c, top;
    stack_init(&st);
    for (int i = 0; s[i] != '\0'; i++) {
        c = s[i];
        if (c == '(' || c == '[' || c == '{') {
            if (stack_push(&st, c) != 0) return 0;    /* 溢出按不匹配处理 */
        } else if (c == ')' || c == ']' || c == '}') {
            if (stack_pop(&st, &top) != 0) return 0;  /* 没有左括号可配 */
            if ((c == ')' && top != '(') || (c == ']' && top != '[') ||
                (c == '}' && top != '{')) return 0;   /* 类型不配对 */
        }
    }
    return stack_empty(&st);         /* 栈空 = 全部配对成功 */
}
int main(void)
{
    printf("%d\n", bracket_match("{[()]}"));  /* 1 */
    printf("%d\n", bracket_match("{[(])}"));  /* 0：交叉不配 */
    printf("%d\n", bracket_match("((("));     /* 0：左括号多余 */
    return 0;
}
```

复杂度与取舍：push/pop 均 O(1)，空间编译期定死、绝不泄漏；代价是超容量按错误处理——这正是"少用堆、多用静态"的缩影。栈深按实际最深嵌套估算并留余量。

测试预期输出：

```
1
0
0
```

## 四、循环队列：数组实现

应用场景一句话：先进先出的任务间数据传递（RTOS 消息队列的雏形）。

```c
#include <stdio.h>
#include <stdint.h>
#define Q_SIZE 8
typedef struct {
    int     buf[Q_SIZE];
    uint8_t front;                   /* 出队端（读） */
    uint8_t rear;                    /* 入队端（写） */
} Queue;

int q_push(Queue *q, int v)          /* 入队：先判满再放数 */
{
    uint8_t next = (uint8_t)((q->rear + 1) % Q_SIZE);
    if (next == q->front) return -1; /* 满：同样牺牲 1 格区分空/满 */
    q->buf[q->rear] = v;
    q->rear = next;
    return 0;
}
int q_pop(Queue *q, int *out)        /* 出队：先判空再取数 */
{
    if (q->front == q->rear) return -1;  /* 空 */
    *out = q->buf[q->front];
    q->front = (uint8_t)((q->front + 1) % Q_SIZE);
    return 0;
}
int main(void)
{
    Queue q = { {0}, 0, 0 };
    for (int i = 1; i <= 7; i++)     /* 实际容量 = Q_SIZE - 1 = 7 */
        if (q_push(&q, i * 10) != 0) printf("push fail\n");
    if (q_push(&q, 80) != 0) printf("full!\n");   /* 第 8 个进不去 */
    int v;
    while (q_pop(&q, &v) == 0) printf("%d ", v);  /* 10 20 30 40 50 60 70 */
    printf("\n");
    return 0;
}
```

与环形缓冲的异同一句话：结构完全同源（数组 + 两个下标 + 留一格判满判空），区别只在用法与命名——环形缓冲强调"中断单写 + 主循环单读"的无锁安全版（head/tail），循环队列是它的通用教科书名字（front/rear），看代码先认清"谁是读端、谁是写端"。

复杂度与取舍：入队出队 O(1)；用"取模循环推进"解决普通数组的假溢出（前部空着却报满）问题。

测试预期输出：

```
full!
10 20 30 40 50 60 70
```

## 五、简单哈希表：字符串命令表（CLI / AT 指令）

应用场景一句话：串口 CLI 把 "help"、"reboot" 这类命令名映射到处理函数——命令上百条时比挨个 strcmp 快（笔记10 第 7 节命令表的进阶版）。

```c
#include <stdio.h>
#include <stdint.h>
#include <string.h>
#define HASH_SIZE 8                 /* 桶数：真实项目取大于条目数的素数更散 */
#define POOL_MAX  16                /* 静态节点池上限 */
typedef void (*cmd_fn)(void);       /* 命令处理函数指针 */
typedef struct Entry {              /* 链地址法：每个桶挂一条单链表 */
    const char   *key;              /* 命令名 */
    cmd_fn        fn;               /* 处理函数 */
    struct Entry *next;             /* 冲突时串成链 */
} Entry;

static Entry *g_buckets[HASH_SIZE]; /* 桶数组：静态分配，嵌入式不用 malloc */
static Entry  g_pool[POOL_MAX];     /* 节点池：预分配（嵌入式取舍） */
static int    g_pool_used = 0;

static unsigned hash_str(const char *s)   /* 哈希：字符加权和取余，简单够用 */
{
    unsigned h = 0;
    while (*s) h = h * 31u + (unsigned char)*s++;
    return h % HASH_SIZE;               /* 真实库用 djb2/FNV 更均匀 */
}
int hash_put(const char *key, cmd_fn fn)  /* 注册命令；重名或池满返回 -1 */
{
    if (g_pool_used >= POOL_MAX) return -1;
    unsigned idx = hash_str(key);
    for (Entry *e = g_buckets[idx]; e != NULL; e = e->next)
        if (strcmp(e->key, key) == 0) return -1;       /* 已存在 */
    Entry *e = &g_pool[g_pool_used++];  /* 从静态池取节点，无需 malloc */
    e->key = key; e->fn = fn;
    e->next = g_buckets[idx];           /* 头插进桶链（复用笔记11 头插法） */
    g_buckets[idx] = e;
    return 0;
}
cmd_fn hash_get(const char *key)          /* 查命令：O(1) 定位桶 + 遍历短链 */
{
    for (Entry *e = g_buckets[hash_str(key)]; e != NULL; e = e->next)
        if (strcmp(e->key, key) == 0) return e->fn;
    return NULL;
}
static void cmd_help(void)    { printf("help: print commands\n"); }
static void cmd_version(void) { printf("ver 1.0\n"); }
static void cmd_reboot(void)  { printf("rebooting...\n"); }
int main(void)
{
    hash_put("help", cmd_help);         /* 注册命令表 */
    hash_put("version", cmd_version);
    hash_put("reboot", cmd_reboot);
    const char *input[] = { "version", "help", "badcmd" };   /* 模拟 CLI 输入 */
    for (int i = 0; i < 3; i++) {
        cmd_fn fn = hash_get(input[i]);
        if (fn != NULL) fn();
        else            printf("unknown: %s\n", input[i]);
    }
    return 0;
}
```

复杂度与取舍：put/get 平均 O(1)，最坏 O(n)（全部撞进同一桶串成链），哈希越散链越短。嵌入式取舍：节点用静态池替代 malloc，容量编译期定死；命令只有十几条时直接用笔记10 的"命令表数组 + 顺序查找"更省 ROM——哈希是命令规模变大后的优化手段。

测试预期输出：

```
ver 1.0
help: print commands
unknown: badcmd
```

## 与 RTOS 的对照表

| 本文实现 | RTOS / 内核对应物 | 一句话关系 |
|---------|------------------|-----------|
| 单链表 | FreeRTOS 就绪列表（list.c，旧类型名 xList） | 任务控制块内嵌链表节点挂队；Linux 内核 list_head 同思想（笔记11 第 6 节） |
| 环形缓冲区 | 串口驱动收发缓冲；xQueue 的底层拷贝区 | 本文是单写单读无锁版；xQueue 在此之上加了阻塞与任务唤醒 |
| 顺序栈 | 每个任务自己的任务栈 | 保存现场与局部变量；栈溢出是嵌入式头号崩溃原因 |
| 循环队列 | xQueue（FreeRTOS 消息队列） | 本质 = 加锁 + 阻塞唤醒的循环队列 |
| 哈希命令表 | FreeRTOS+CLI、各类嵌入式 shell | 命令注册与查找 |
| 静态节点池 | heap_4（带合并算法的内存池方案） | 裸机尽量全静态；RTOS 下经 pvPortMalloc 走 heap_4（笔记08 第 4 节） |
